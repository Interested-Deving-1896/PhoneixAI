async chat(messages, provider, model, config) {
const startTime = Date.now();

```
try {
  // Try primary provider first
  const response = await this.makeAPICall(messages, provider, model, config);
  return {
    ...response,
    metadata: {
      ...response.metadata,
      responseTime: Date.now() - startTime,
      provider,
      model
    }
  };
} catch (error) {
  console.warn(`Provider ${provider} failed:`, error.message);
  
  // Only try fallbacks if enabled
  if (!config.fallbackEnabled) {
    throw error;
  }
  
  // Try fallback providers
  for (const fallbackProvider of this.fallbackOrder) {
    if (fallbackProvider === provider) continue;
    
    try {
      await new Promise(resolve => setTimeout(resolve, this.retryDelay));
      const fallbackModel = aiProviders[fallbackProvider]?.models[0]?.id;
      if (!fallbackModel) continue;
      
      const response = await this.makeAPICall(messages, fallbackProvider, fallbackModel, config);
      return {
        ...response,
        metadata: {
          ...response.metadata,
          responseTime: Date.now() - startTime,
          provider: fallbackProvider,
          model: fallbackModel,
          fallbackUsed: true,
          originalProvider: provider
        }
      };
    } catch (fallbackError) {
      console.warn(`Fallback provider ${fallbackProvider} also failed:`, fallbackError.message);
    }
  }
  
  throw new Error(`All providers failed. Original error: ${error.message}`);
}
```

}
