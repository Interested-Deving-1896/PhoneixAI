// lib/ai/providers/base.ts
export interface AIMessage {
role: ‘user’ | ‘assistant’ | ‘system’;
content: string;
timestamp?: Date;
}

export interface AIResponse {
content: string;
model: string;
provider: string;
metadata: {
tokensUsed?: number;
responseTime: number;
confidence?: number;
finishReason?: string;
};
}

export interface AIConfig {
temperature: number;
maxTokens: number;
streamEnabled: boolean;
systemPrompt?: string;
}

export interface ProviderConfig {
apiKey: string;
baseUrl?: string;
organization?: string;
}

export abstract class BaseAIProvider {
protected config: ProviderConfig;
protected name: string;

constructor(config: ProviderConfig, name: string) {
this.config = config;
this.name = name;
}

abstract chat(
messages: AIMessage[],
model: string,
config: AIConfig
): Promise<AIResponse>;

abstract streamChat(
messages: AIMessage[],
model: string,
config: AIConfig,
onChunk: (chunk: string) => void
): Promise<AIResponse>;

abstract getModels(): Promise<string[]>;
abstract checkHealth(): Promise<boolean>;

getName(): string {
return this.name;
}
}

async streamChat(
messages: AIMessage[],
model: string,
config: AIConfig,
onChunk: (chunk: string) => void
): Promise<AIResponse> {
const startTime = Date.now();

```
try {
  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': this.config.apiKey,
      'anthropic-version': '2023-06-01',
    },
    body: JSON.stringify({
      model,
      max_tokens: config.maxTokens,
      temperature: config.temperature,
      messages: messages.filter(m => m.role !== 'system'),
      system: config.systemPrompt || messages.find(m => m.role === 'system')?.content,
      stream: true,
    }),
  });

  if (!response.ok) {
    throw new Error(`Anthropic API error: ${response.status}`);
  }

  const reader = response.body?.getReader();
  const decoder = new TextDecoder();
  let fullContent = '';

  if (reader) {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const chunk = decoder.decode(value);
      const lines = chunk.split('\n');

      for (const line of lines) {
        if (line.startsWith('data: ')) {
          try {
            const data = JSON.parse(line.slice(6));
            if (data.type === 'content_block_delta') {
              const text = data.delta?.text || '';
              fullContent += text;
              onChunk(text);
            }
          } catch (e) {
            // Skip invalid JSON
          }
        }
      }
    }
  }

  const responseTime = Date.now() - startTime;

  return {
    content: fullContent,
    model,
    provider: this.name,
    metadata: {
      responseTime,
      finishReason: 'stream_complete',
    },
  };
} catch (error) {
  throw new Error(`Anthropic streaming error: ${error.message}`);
}
```

}

async streamChat(
messages: AIMessage[],
model: string,
config: AIConfig,
onChunk: (chunk: string) => void
): Promise<AIResponse> {
const startTime = Date.now();

```
try {
  const response = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${this.config.apiKey}`,
    },
    body: JSON.stringify({
      model,
      messages: messages.map(m => ({ role: m.role, content: m.content })),
      temperature: config.temperature,
      max_tokens: config.maxTokens,
      stream: true,
    }),
  });

  if (!response.ok) {
    throw new Error(`OpenAI API error: ${response.status}`);
  }

  const reader = response.body?.getReader();
  const decoder = new TextDecoder();
  let fullContent = '';

  if (reader) {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const chunk = decoder.decode(value);
      const lines = chunk.split('\n');

      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = line.slice(6).trim();
          if (data === '[DONE]') break;

          try {
            const parsed = JSON.parse(data);
            const content = parsed.choices[0]?.delta?.content || '';
            if (content) {
              fullContent += content;
              onChunk(content);
            }
          } catch (e) {
            // Skip invalid JSON
          }
        }
      }
    }
  }

  const responseTime = Date.now() - startTime;

  return {
    content: fullContent,
    model,
    provider: this.name,
    metadata: {
      responseTime,
      finishReason: 'stream_complete',
    },
  };
} catch (error) {
  throw new Error(`OpenAI streaming error: ${error.message}`);
}

// Initialize providers from configuration
async initializeProviders(configs: Record<string, ProviderConfig>): Promise<void> {
const providerClasses = {
anthropic: AnthropicProvider,
openai: OpenAIProvider,
google: GoogleProvider,
cohere: CohereProvider,
local: LocalProvider,
};

```
for (const [name, config] of Object.entries(configs)) {
  if (config.apiKey || name === 'local') {
    const ProviderClass = providerClasses[name as keyof typeof providerClasses];
    if (ProviderClass) {
      const provider = new ProviderClass(config);
      this.providers.set(name, provider);

      // Initialize provider settings
      this.providerSettings.set(name, {
        name,
        enabled: true,
        priority: this.fallbackConfig.fallbackOrder.indexOf(name),

      // Get available models
      try {
        const models = await provider.getModels();
        const settings = this.providerSettings.get(name)!;
        settings.models = models;
        this.providerSettings.set(name, settings);
      } catch (error) {
        console.warn(`Failed to get models for ${name}:`, error);
      }
    }
  }
}

// Initial health check
await this.checkAllProviders();

// Main chat method with automatic fallback
async chat(
messages: AIMessage[],
preferredProvider: string,
model: string,
config: AIConfig
): Promise<AIResponse> {
const providers = this.getOrderedProviders(preferredProvider);
let lastError: Error | null = null;

```
for (const providerName of providers) {
  const provider = this.providers.get(providerName);
  const settings = this.providerSettings.get(providerName);

  if (!provider || !settings?.enabled || settings.status === 'offline') {
    continue;
  }

  try {
    console.log(`Attempting chat with ${providerName}...`);
    const response = await provider.chat(messages, model, config);
    
    // Update status on success
    this.updateProviderStatus(providerName, 'online');
    
    return response;
  } catch (error) {
    console.warn(`Provider ${providerName} failed:`, error);
    lastError = error as Error;
    
    // Update status on failure
    this.updateProviderStatus(providerName, 'offline');
    
    if (!this.fallbackConfig.enabled) {
      throw error;
    }

    // Wait before trying next provider
    if (this.fallbackConfig.retryDelay > 0) {
      await new Promise(resolve => setTimeout(resolve, this.fallbackConfig.retryDelay));
    }
  }
}

throw new Error(
  `All providers failed. Last error: ${lastError?.message || 'Unknown error'}`
);
```

}

              const data = JSON.parse(line.slice(6));
              if (data.type === 'chunk') {
                fullContent += data.content;
                setStreamingContent(fullContent);
              } else if (data.type === 'complete') {
                const assistantMessage: ChatMessage = {
                  id: (Date.now() + 1).toString(),
                  type: 'assistant',
                  content: data.response.content,
                  timestamp: new Date(),
                  model: `${selectedProvider}/${selectedModel}`,
                  metadata: data.response.metadata
                };
                setMessages(prev => [...prev, assistantMessage]);
                setStreamingContent('');
                setIsLoading(false);
                return;
              } else if (data.type === 'error') {
                throw new Error(data.error);
              }
            } catch (e) {
              // Skip invalid JSON
            }
          }
        }
      }
    }
  } else {
    // Regular response
    const response = await fetch('/api/ai/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },

        messages: aiMessages,

        config: aiConfig
