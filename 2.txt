// lib/middleware/auth.ts
import { NextRequest, NextResponse } from ‘next/server’;
import { getToken } from ‘next-auth/jwt’;

export async function authMiddleware(
request: NextRequest,
response: NextResponse
) {
const token = await getToken({ req: request });

// Protected API routes
if (request.nextUrl.pathname.startsWith(’/api/ai/’)) {
if (!token) {
return NextResponse.json(
{ error: ‘Authentication required’ },
{ status: 401 }
);
}
}

return response;
}

// lib/middleware/rateLimit.ts
import { NextRequest, NextResponse } from ‘next/server’;

interface RateLimitStore {
[key: string]: {
count: number;
resetTime: number;
};
}

const store: RateLimitStore = {};

export function rateLimit(
identifier: string,
limit: number = 60,
windowMs: number = 60 * 1000 // 1 minute
): boolean {
const now = Date.now();
const windowStart = now - windowMs;

// Clean up old entries
Object.keys(store).forEach(key => {
if (store[key].resetTime < windowStart) {
delete store[key];
}
});

if (!store[identifier]) {
store[identifier] = {
count: 1,
resetTime: now + windowMs,
};
return true;
}

if (store[identifier].resetTime < now) {
store[identifier] = {
count: 1,
resetTime: now + windowMs,
};
return true;
}

if (store[identifier].count >= limit) {
return false;
}

store[identifier].count++;
return true;
}

export async function rateLimitMiddleware(request: NextRequest) {
const ip = request.ip || request.headers.get(‘x-forwarded-for’) || ‘unknown’;
const identifier = `${ip}-${request.nextUrl.pathname}`;

const isAllowed = rateLimit(
identifier,
parseInt(process.env.RATE_LIMIT_REQUESTS_PER_MINUTE || ‘60’),
60 * 1000
);

if (!isAllowed) {
return NextResponse.json(
{ error: ‘Rate limit exceeded’ },
{ status: 429 }
);
}

return null; // Continue
}

FROM base AS builder
WORKDIR /app
COPY –from=deps /app/node_modules ./node_modules
COPY . .

# Generate Prisma client

RUN npx prisma generate

# Build the application

RUN npm run build

# Production image, copy all the files and run next

FROM base AS runner
WORKDIR /app

ENV NODE_ENV production

RUN addgroup –system –gid 1001 nodejs
RUN adduser –system –uid 1001 nextjs

COPY –from=builder /app/public ./public

# Set the correct permission for prerender cache

RUN mkdir .next
RUN chown nextjs:nodejs .next

# Automatically leverage output traces to reduce image size

COPY –from=builder –chown=nextjs:nodejs /app/.next/standalone ./
COPY –from=builder –chown=nextjs:nodejs /app/.next/static ./.next/static

USER nextjs

EXPOSE 3000

ENV PORT 3000
ENV HOSTNAME “0.0.0.0”

CMD [“node”, “server.js”]

# scripts/setup.js

#!/usr/bin/env node

const fs = require(‘fs’);
const path = require(‘path’);
const crypto = require(‘crypto’);

console.log(‘🚀 Setting up Phoenix AI Universal…\n’);

// Generate secure secrets
function generateSecret() {
return crypto.randomBytes(32).toString(‘hex’);
}

// Check if .env.local exists
const envPath = path.join(process.cwd(), ‘.env.local’);
const envExamplePath = path.join(process.cwd(), ‘.env.example’);

if (!fs.existsSync(envPath)) {
if (fs.existsSync(envExamplePath)) {
// Copy .env.example to .env.local
let envContent = fs.readFileSync(envExamplePath, ‘utf8’);

```
// Generate and replace secrets
envContent = envContent.replace(
  'NEXTAUTH_SECRET="your-super-secret-jwt-secret-here"',
  `NEXTAUTH_SECRET="${generateSecret()}"`
);

fs.writeFileSync(envPath, envContent);
console.log('✅ Created .env.local from .env.example');
console.log('⚠️  Please update the API keys in .env.local');
```

} else {
console.log(‘❌ .env.example not found’);
process.exit(1);
}
} else {
console.log(‘✅ .env.local already exists’);
}

// Check for required tools
console.log(’\n🔍 Checking system requirements…’);

const requiredCommands = [‘node’, ‘npm’, ‘npx’];
for (const cmd of requiredCommands) {
try {
require(‘child_process’).execSync(`${cmd} --version`, { stdio: ‘ignore’ });
console.log(`✅ ${cmd} is installed`);
} catch (error) {

# scripts/deploy.js

#!/usr/bin/env node

const { execSync } = require(‘child_process’);

const platform = process.argv[2];

if (!platform) {
console.log(‘Usage: npm run deploy <platform>’);
console.log(‘Platforms: vercel, railway, render, docker’);
process.exit(1);
}

console.log(`🚀 Deploying to ${platform}...`);

switch (platform) {
case ‘vercel’:
execSync(‘vercel –prod’, { stdio: ‘inherit’ });
break;

case ‘railway’:
execSync(‘railway up’, { stdio: ‘inherit’ });
break;

case ‘render’:
console.log(‘Please push to your connected repository for Render deployment’);
break;

case ‘docker’:
execSync(‘docker build -t phoenix-ai .’, { stdio: ‘inherit’ });
execSync(‘docker-compose up -d’, { stdio: ‘inherit’ });
break;

default:
console.log(`Unknown platform: ${platform}`);
process.exit(1);
}

console.log(`✅ Deployment to ${platform} initiated!`);

# README.md

# 🔥 Phoenix AI Universal

> **The Ultimate Universal AI Interface** - Compatible with ANY AI model provider

Phoenix AI is a sophisticated, production-ready interface that provides seamless access to multiple AI providers through a single, elegant platform. Built with enterprise-grade architecture, it offers universal compatibility, intelligent fallbacks, and a premium user experience.

## ✨ Features

### 🌐 Universal AI Provider Support

- **Anthropic** (Claude 3.5 Sonnet, Haiku, Opus)
- **OpenAI** (GPT-4, GPT-3.5 Turbo, GPT-4 Turbo)
- **Google** (Gemini Pro, Gemini Flash)
- **Cohere** (Command R+, Command R)
- **Local Models** (Ollama, LMStudio, any OpenAI-compatible API)

### 🚀 Production-Ready Architecture

- **Intelligent Fallbacks** - Automatic provider switching on failures
- **Real-time Streaming** - Live response streaming from all providers
- **Advanced Configuration** - Temperature, tokens, custom prompts
- **Rate Limiting** - Built-in protection and usage tracking
- **Authentication** - NextAuth.js with multiple providers

### 🎨 Premium User Experience

- **Phoenix Workflow** - Structured development phases
- **Modern UI** - 2025 design trends with accessibility
- **Export Options** - Markdown, JSON, PDF conversation exports
- **Session Management** - Persistent chat history
- **Responsive Design** - Perfect on all devices

## 🚀 Quick Start

### Prerequisites

- Node.js 18+
- PostgreSQL (or SQLite for development)
- At least one AI provider API key

### 1. Clone & Setup

```bash
git clone https://github.com/your-org/phoenix-ai-universal.git
cd phoenix-ai-universal
npm install
npm run setup
```

### 2. Configure Environment

Copy `.env.example` to `.env.local` and add your API keys:

```bash
# Required: At least one AI provider
ANTHROPIC_API_KEY="sk-ant-your-key-here"
OPENAI_API_KEY="sk-your-openai-key-here"
GOOGLE_API_KEY="your-google-api-key-here"

# Database
DATABASE_URL="postgresql://user:password@localhost:5432/phoenix_ai"

# Authentication
NEXTAUTH_SECRET="auto-generated-secure-secret"
NEXTAUTH_URL="http://localhost:3000"
```

### 3. Initialize Database

```bash
npx prisma db push
npx prisma generate
```

### 4. Start Development

```bash
npm run dev
```

Open <http://localhost:3000> 🎉
